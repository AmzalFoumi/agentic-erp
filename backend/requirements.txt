# Python dependencies for the backend.
#
# This file is the *recipe* for the virtual environment. It is committed to git;
# the environment it builds (.venv/) is not. Anyone cloning this repo runs
# `pip install -r requirements.txt` and gets the exact same set of packages.
#
# `==` pins an exact version. We pin everything so that a build today and a
# build in six months are identical - the single most common source of
# "it works on my machine" is an unpinned dependency quietly upgrading.
#
# Verified against PyPI on 2026-07-29.

# ---------- Web framework (the api/ adapter) ----------

# FastAPI turns plain Python functions into HTTP endpoints. It reads your type
# hints at runtime to validate incoming JSON and to generate the interactive
# docs at /docs. Requires Python 3.10+; we are on 3.12.4.
fastapi==0.141.0

# The actual web server that runs the FastAPI app. FastAPI describes the app;
# uvicorn is the process that listens on a port. [standard] pulls in optional
# speed-ups and the file-watcher that powers `--reload`.
uvicorn[standard]==0.52.0

# ---------- Database (core/) ----------

# SQLAlchemy is the ORM: it maps Python classes to SQL tables, so we write
# `session.get(Product, 1)` instead of hand-writing SELECT statements.
# The 2.x style is quite different from 1.x - when reading tutorials online,
# check they are 2.0-era.
sqlalchemy==2.0.51

# The Postgres driver. SQLAlchemy speaks SQL; psycopg speaks to the actual
# server over the network. [binary] ships precompiled, so no C compiler is
# needed on Windows.
psycopg[binary]==3.3.4

# Alembic generates and applies database migrations - versioned, ordered
# scripts that evolve the schema. Written by the SQLAlchemy author.
alembic==1.18.5

# ---------- Configuration ----------

# Reads settings from a .env file into a typed Python object, so a missing or
# malformed setting fails loudly at startup rather than mysteriously later.
pydantic-settings==2.14.2

# ---------- MCP (the mcp_server/ adapter) ----------

# The MCP Python SDK - exposes our service functions as tools an AI agent can
# call. Server class is `MCPServer` (older tutorials show `FastMCP`).
#
# NOTE: version 2.0.0 came out 2026-07-28, brand new at the time we picked it,
# and is a big rework (drops the old handshake/session setup for a simpler,
# stateless one). We chose it on purpose to match current docs, not by
# accident. If it turns out too rough to work with, fall back to the older,
# stable version: mcp==1.29.0
mcp==2.0.0

# ---------- Testing ----------

# The test runner. Finds files named test_*.py and functions named test_*.
pytest==9.1.1

# An HTTP client. FastAPI's test client is built on it, so it is what we use to
# call our own API from inside a test without starting a real server.
httpx==0.28.1

# ---------- Architecture enforcement ----------

# Enforces the services/ boundary rule mechanically instead of by memory.
# Adoption reasoning and config history: docs/BACKEND-PLAN.md, Gate 4.
import-linter==2.13

# ---------- Authentication (the authn/ package, gate 24) ----------

# Verifies the access tokens ThunderID issues: signature, issuer, audience,
# expiry. [crypto] pulls in `cryptography`, which supplies the RSA primitives -
# without it pyjwt can only do HMAC and RS256 verification fails at import.
#
# Both were already present as transitive dependencies before gate 24. They are
# pinned explicitly now because we import them directly, and a package you
# import is a package you own the version of.
#
# ⚠️ `jwt` is also listed in BOTH forbidden_modules contracts in pyproject.toml.
# Adding the dependency without that line would let services/ import it and the
# architecture boundary would weaken silently. Verified against PyPI 2026-08-23.
pyjwt[crypto]==2.13.0
